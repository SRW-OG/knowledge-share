# 本地化 AI 记忆系统架构与部署指南

> **最新更新：2026-02-27** — 针对低配电脑的轻量化优化方案

---

## 1. 问题现状与架构演进

### 背景

在 AI Agent 系统中，如何让 AI 具备持久记忆是一个核心挑战。本文档记录了从"重服务架构"到"轻量化架构"的演进过程。

### 架构演进

| 阶段 | 时间 | 架构 | 内存占用 |
|------|------|------|----------|
| **v1.0** | 2026-02 | Qdrant + Redis + PostgreSQL | ~1GB+ |
| **v2.0** (当前) | 2026-02-27 | FAISS + SQLite + PostgreSQL | ~100MB |

### 优化背景

- **Qdrant**：作为独立 Docker 服务，500条索引完全不需要，启动即占几百MB内存
- **Redis**：仅做缓存时，Python dict 或 SQLite 性能足够，且不占常驻内存

---

## 2. 新架构：轻量级"双轨制"记忆系统

###.1 核心 2设计理念

**本地承担存储与检索，云端免费 API 承担语义压缩**

### 2.2 三层存储结构

```
┌─────────────────────────────────────────────────────────────────┐
│                      用户提问                                    │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                    感知层 (Sensory Memory)                       │
│  当前对话最后 2-3 轮原始文本 (内存)                              │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                  短程层 (Episodic Memory - SQLite)               │
│  索引: BM25 + 时间戳                                            │
│  内容: 最近 50 轮对话摘要                                       │
└─────────────────────────────────────────────────────────────────┘
                              │
                              ▼
┌─────────────────────────────────────────────────────────────────┐
│                  长程层 (Semantic Memory - FAISS)                 │
│  索引: FAISS + BGE-Small-ZH                                    │
│  内容: 提取的事实 (Fact) 和用户偏好                              │
└─────────────────────────────────────────────────────────────────┘
                              │
            ┌─────────────────┼─────────────────┐
            ▼                 ▼                 ▼
    ┌──────────────┐  ┌──────────────┐  ┌──────────────┐
    │  智谱 AI     │  │  通义千问     │  │  BGE-Small  │
    │ (免费 API)   │  │ (免费 API)   │  │  (本地向量)  │
    └──────────────┘  └──────────────┘  └──────────────┘
```

### 2.3 技术栈选型

| 组件 | 选型 | 理由 |
|------|------|------|
| **LLM API** | **智谱 AI (GLM-4-Flash)** 或 **通义千问** | 注册送大量 Token，国内访问稳定，免费 |
| **Embedding** | **BGE-Small-ZH** | 仅约 100MB，CPU 快速运行，无需 GPU |
| **数据库** | **SQLite** | Python 自带，零开销 |
| **检索增强** | **Rank-BM25** | 纯 Python，关键词匹配更快更省内存 |
| **向量检索** | **FAISS (CPU版)** | 本地文件，平时不占内存 |

### 4.3 建立本地向量索引

创建 `sync_memory.py` 脚本：

```python
import os
from sentence_transformers import SentenceTransformer
import faiss
import numpy as np
import json

# 使用国内可直接下载的极小模型 (约100MB)
MODEL_NAME = 'BAAI/bge-small-zh-v1.5'

def build_index():
    # 1. 读取所有 .md 文件
    docs = []
    paths = []
    memory_dir = os.path.expanduser("~/.openclaw/workspace/memory/")
    
    for root, dirs, files in os.walk(memory_dir):
        for file in files:
            if file.endswith(".md"):
                path = os.path.join(root, file)
                with open(path, 'r', encoding='utf-8') as f:
                    content = f.read()
                    docs.append(content)
                    paths.append(path)
    
    if not docs:
        print("No markdown files found")
        return
    
    print(f"Processing {len(docs)} files...")
    
    # 2. 生成向量 (CPU 上很快)
    model = SentenceTransformer(MODEL_NAME)
    embeddings = model.encode(docs, show_progress_bar=True)
    
    # 3. 保存为本地 FAISS 索引
    dimension = embeddings.shape[1]
    index = faiss.IndexFlatL2(dimension)
    index.add(np.array(embeddings))
    
    # 保存索引文件
    faiss.write_index(index, "memory_index.faiss")
    
    # 保存文档映射
    with open("docs_mapping.json", "w", encoding='utf-8') as f:
        json.dump({"paths": paths, "docs": docs}, f, ensure_ascii=False)
    
    print(f"Index saved: memory_index.faiss ({os.path.getsize('memory_index.faiss')/1024/1024:.2f}MB)")

if __name__ == "__main__":
    build_index()
```

### 4.4 搜索逻辑

```python
import faiss
import numpy as np
import json
from sentence_transformers import SentenceTransformer

MODEL_NAME = 'BAAI/bge-small-zh-v1.5'

def search_memory(query, top_k=3):
    # 加载索引
    index = faiss.read_index("memory_index.faiss")
    with open("docs_mapping.json", "r", encoding='utf-8') as f:
        data = json.load(f)
    
    # 向量搜索
    model = SentenceTransformer(MODEL_NAME)
    query_vec = model.encode([query])
    
    distances, indices = index.search(query_vec, top_k)
    
    # 返回结果
    results = []
    for i, idx in enumerate(indices[0]):
        if idx < len(data["paths"]):
            results.append({
                "path": data["paths"][idx],
                "content": data["docs"][idx][:500],  # 截取前500字符
                "distance": distances[0][i]
            })
    
    return results

# 测试
if __name__ == "__main__":
    results = search_memory("我的项目进展")
    for r in results:
        print(f"- {r['path']}: {r['distance']:.4f}")
```

### 4.5 混合检索逻辑（推荐）

```python
import os
import re
from rank_bm25 import BM25Okapi
import faiss
import numpy as np
import json
from sentence_transformers import SentenceTransformer

MODEL_NAME = 'BAAI/bge-small-zh-v1.5'

def hybrid_search(query, top_k=3):
    memory_dir = os.path.expanduser("~/.openclaw/workspace/memory/")
    
    # 加载文档
    docs = []
    paths = []
    for root, dirs, files in os.walk(memory_dir):
        for file in files:
            if file.endswith(".md"):
                path = os.path.join(root, file)
                with open(path, 'r', encoding='utf-8') as f:
                    docs.append(f.read())
                    paths.append(path)
    
    if not docs:
        return []
    
    # 步骤 A: BM25 关键词搜索
    tokenized = [doc.lower().split() for doc in docs]
    bm25 = BM25Okapi(tokenized)
    bm25_results = bm25.get_scores(query.lower().split())
    
    # 获取 BM25 top 结果
    bm25_top = sorted(range(len(bm25_results)), 
                     key=lambda i: bm25_results[i], 
                     reverse=True)[:top_k]
    
    # 步骤 B: 如需语义补充，加载 FAISS
    try:
        index = faiss.read_index("memory_index.faiss")
        with open("docs_mapping.json", "r", encoding='utf-8') as f:
            data = json.load(f)
        
        model = SentenceTransformer(MODEL_NAME)
        query_vec = model.encode([query])
        _, faiss_indices = index.search(query_vec, top_k)
        
        # 合并结果（去重 + 排序）
        seen = set()
        final_results = []
        for idx in list(bm25_top) + list(faiss_indices[0]):
            if idx not in seen and idx < len(paths):
                seen.add(idx)
                final_results.append({
                    "path": paths[idx],
                    "content": docs[idx][:300],
                    "score": bm25_results[idx]
                })
                if len(final_results) >= top_k:
                    break
        
        return final_results
    except:
        # 如果 FAISS 失败，返回 BM25 结果
        return [{"path": paths[i], "content": docs[i][:300], "score": bm25_results[i]} 
                for i in bm25_top[:top_k]]
```

### 5.2 每周任务脚本

```python
import requests
import os
from datetime import datetime, timedelta

ZHIPU_API_KEY = "your-api-key"

def extract_weekly_memory():
    # 1. 收集本周日志
    memory_dir = os.path.expanduser("~/.openclaw/workspace/memory/")
    week_ago = (datetime.now() - timedelta(days=7)).strftime("%Y-%m-%d")
    
    content = []
    for file in os.listdir(memory_dir):
        if file >= week_ago and file.endswith(".md"):
            with open(os.path.join(memory_dir, file), 'r') as f:
                content.append(f.read())
    
    if not content:
        print("No files to process")
        return
    
    combined = "\n\n---\n\n".join(content)
    
    # 2. 调用智谱 API 提炼
    prompt = f"""以下是本周的日志，请提取其中的重要事实：
    - 人名和关系
    - 项目进度
    - 用户偏好
    - 关键决策
    
    请以简洁的 Markdown 格式输出：
    
    {combined[:5000]}"""  # 限制长度
    
    response = requests.post(
        "https://open.bigmodel.cn/api/paas/v4/chat/completions",
        headers={"Authorization": f"Bearer {ZHIPU_API_KEY}"},
        json={"model": "glm-4-flash", "messages": [{"role": "user", "content": prompt}]}
    )
    
    result = response.json()['choices'][0]['message']['content']
    
    # 3. 更新 MEMORY.md
    memory_file = os.path.expanduser("~/.openclaw/workspace/MEMORY.md")
    with open(memory_file, 'a', encoding='utf-8') as f:
        f.write(f"\n\n## {datetime.now().strftime('%Y-%m-%d')} 周总结\n\n")
        f.write(result)
    
    print("MEMORY.md updated")

if __name__ == "__main__":
    extract_weekly_memory()
```

## 6. 安全过滤：三层"防火墙"保护隐私

> **重要安全提醒**：当将搜索到的记忆片段发送给云端 API 时，AI 会接收到你发送的所有文本。如果包含 API Key、密码或私钥，它们就会被上传到云端。

### 6.1 问题背景

| 风险 | 说明 |
|------|------|
| API Key 泄露 | memory 文件中可能包含调试时输入的 Key |
| 密码泄露 | 数据库密码、服务器密码等 |
| Token 泄露 | 各种长度的十六进制字符串 |

### 6.2 三层防火墙

#### 第一层：本地自动化脱敏 (Regex Filtering)

在将任何文本发送给云端 API 之前，运行正则表达式过滤：

```python
import re

def sanitize_text(text):
    """在发送到云端前，擦除所有敏感信息"""
    patterns = {
        "ZHIPU_KEY": r'[a-zA-Z0-9]{32}\.[a-zA-Z0-9]{16}',
        "GENERIC_KEY": r'sk-[a-zA-Z0-9]{32,}',
        "DB_URL": r'postgresql://\w+:\w+@[\w\.-]+:\d+/\w+',
        "PASSWORD": r'(password|pwd|secret)[:=]\s*[^\s]+',
        "HEX_SECRET": r'([0-9a-fA-F]{32,})',
    }
    
    clean_text = text
    for label, pattern in patterns.items():
        clean_text = re.sub(pattern, f"[{label}_REDACTED]", clean_text)
    
    return clean_text

# 使用示例
safe_context = sanitize_text(original_context)
safe_query = sanitize_text(original_query)
```

#### 第二层：检索源头控制 (Ignore List)

在建立索引时，过滤敏感文件：


**最佳实践**：
- API Key 存放在 `~/.openclaw/` 根目录，不要放在 `memory/` 下
- Obsidian 知识库中创建 `/public_notes/` 子文件夹专门给 OpenClaw 读取

#### 第三层：结构化提炼 (Fact Memory)

- **原始数据**（本地）：`memory/2026-02-27.md` 包含原始对话
- **提炼数据**：`MEMORY.md` 只记录精简的事实


## 7. 完整示例：本地检索 + 自动脱敏 + 智谱调用

```python
import os
import re
import json
import requests
import numpy as np
import faiss
from rank_bm25 import BM25Okapi
from sentence_transformers import SentenceTransformer

# ================= 配置区 =================
ZHIPU_API_KEY = "your-api-key"  # 替换为你的 API Key
MEMORY_PATH = os.path.expanduser("~/.openclaw/workspace/memory/")
MODEL_NAME = 'BAAI/bge-small-zh-v1.5'

# ================= 脱敏工具 =================
def sanitize_text(text):
    """在发送到云端前，擦除所有敏感信息"""
    patterns = {
        "ZHIPU_KEY": r'[a-zA-Z0-9]{32}\.[a-zA-Z0-9]{16}',
        "OPENAI_KEY": r'sk-[a-zA-Z0-9]{32,}',
        "DB_URL": r'postgresql://\w+:\w+@[\w\.-]+:\d+/\w+',
        "PASSWORD": r'(password|pwd|secret)[:=]\s*[^\s]+',
        "HEX_SECRET": r'([0-9a-fA-F]{32,})',
    }
    clean_text = text
    for label, pattern in patterns.items():
        clean_text = re.sub(pattern, f"[{label}_REDACTED]", clean_text)
    return clean_text

# ================= 本地混合搜索类 =================
class LocalMemoryEngine:
    def __init__(self):
        self.docs = []
        self.paths = []
        self.model = None
        self.index = None
        self.load_docs()
    
    def load_docs(self):
        """加载所有 Markdown 记忆"""
        for root, dirs, files in os.walk(MEMORY_PATH):
            for file in files:
                if file.endswith(".md"):
                    path = os.path.join(root, file)
                    with open(path, 'r', encoding='utf-8') as f:
                        content = f.read()
                        # 按段落拆分
                        paragraphs = [p.strip() for p in content.split('\n\n') if len(p) > 10]
                        self.docs.extend(paragraphs)
                        self.paths.extend([path] * len(paragraphs))
    
    def search(self, query, top_k=3):
        """混合检索：BM25 + FAISS"""
        if not self.docs:
            return []
        
        # 1. BM25 关键词检索
        tokenized = [list(doc) for doc in self.docs]
        bm25 = BM25Okapi(tokenized)
        bm25_scores = bm25.get_scores(list(query))
        
        # 获取 top 结果
        bm25_top = sorted(range(len(bm25_scores)), 
                         key=lambda i: bm25_scores[i], 
                         reverse=True)[:top_k]
        
        # 2. 语义检索（如需要）
        try:
            if self.model is None:
                self.model = SentenceTransformer(MODEL_NAME)
            
            query_vec = self.model.encode([query])
            doc_vecs = self.model.encode(self.docs)
            
            index = faiss.IndexFlatL2(doc_vecs.shape[1])
            index.add(np.array(doc_vecs))
            _, faiss_indices = index.search(query_vec, top_k)
            
            # 合并结果
            seen = set()
            results = []
            for idx in list(bm25_top) + list(faiss_indices[0]):
                if idx not in seen and idx < len(self.docs):
                    seen.add(idx)
                    results.append({
                        "path": self.paths[idx],
                        "content": self.docs[idx][:300]
                    })
                    if len(results) >= top_k:
                        break
            return results
        except Exception as e:
            # 降级返回 BM25 结果
            return [{"path": self.paths[i], "content": self.docs[i][:300]} 
                    for i in bm25_top[:top_k]]

# ================= 调用智谱 API =================
def ask_zhipu(query):
    engine = LocalMemoryEngine()
    
    # 1. 本地检索
    results = engine.search(query)
    context = "\n---\n".join([r['content'] for r in results])
    
    # 2. 脱敏处理（关键！）
    safe_context = sanitize_text(context)
    safe_query = sanitize_text(query)
    
    # 3. 构造 Prompt
    prompt = f"""你是一个拥有记忆的助手。请基于以下【参考记忆】回答用户问题。
如果记忆中没有相关信息，请直接回答。回答中严禁包含任何密钥、密码或敏感字符串。
```
