# 本地文档分类与Markdown转换自动化方案

## 一、问题现状

### 背景描述

用户在构建个人知识库过程中积累了大量的本地文档，包括：
- GB/T、GA/T 等网络安全国标文件
- 等保、密评相关政策文档
- CISSP、CISP 等认证考试教材
- 各类安全报告、白皮书

这些文档格式混杂（PDF、Word、PPT、Excel等），存储混乱，难以进行高效检索和知识管理。

### 影响范围

- 知识检索效率低下
- 重复文档占用存储空间
- 无法利用 AI 进行 RAG（检索增强生成）问答
- 文档复用困难

---

## 二、解决思路与方案对比

### 方案对比

| 方案 | 成本 | 优点 | 缺点 | 适用场景 |
|------|------|------|------|----------|
| 纯本地 Ollama | $0 | 隐私安全、完全离线 | 需要高性能显卡 | 有 GPU 的用户 |
| DeepSeek API | <¥10/千份 | 性价比极高、中文优化 | 需要网络连接 | 普通用户 |
| 纯 MarkItDown | $0 | 免费开源、支持多格式 | 无法智能分类 | 仅格式转换 |

### 推荐方案

采用**"本地预处理 + 廉价大模型 API（DeepSeek）+ 自动化脚本"**的组合，既保证低成本，又具备智能分类能力。

### 核心Python脚本

```python
import os
import shutil
import hashlib
import json
import requests
from markitdown import MarkItDown
import yaml

# ================= 配置区域 =================
CONFIG = {
    "input_dir": r"D:\Downloads\未整理文档", 
    "output_base": r"D:\My_Knowledge_Base",
    "ollama_url": "http://localhost:11434/api/chat",
    "model_name": "qwen3.5:9b",
    "extensions": ['.pdf', '.docx', '.pptx', '.xlsx', '.txt', '.md', '.html']
}

DIR_SOURCE = os.path.join(CONFIG["output_base"], "00_Original_Files")
DIR_MD = os.path.join(CONFIG["output_base"], "01_Obsidian_Vault")

md_converter = MarkItDown()
seen_hashes = set()

def get_file_hash(filepath):
    """计算文件哈希值用于去重"""
    hasher = hashlib.sha256()
    with open(filepath, 'rb') as f:
        while buf := f.read(65536):
            hasher.update(buf)
    return hasher.hexdigest()

def ai_classify(text_snippet):
    """调用 Ollama 对文本进行分类"""
    if not text_snippet or len(text_snippet) < 50:
        return "未分类_其他"

    prompt = f"""
    你是一个专业的文件整理助手。请根据以下文档摘要，给它归类到一个合适的文件夹名称中。
    分类名称格式：一级类目_二级类目（例如：技术_Python、工作_合同）。
    如果无法确定，返回 "杂项"。
    
    文档摘要：
    {text_snippet[:1500]} 
    """

    payload = {
        "model": CONFIG["model_name"],
        "messages": [{"role": "user", "content": prompt}],
        "stream": False
    }

    try:
        response = requests.post(CONFIG["ollama_url"], json=payload)
        return response.json()['message']['content'].strip().split('\n')[0]
    except Exception as e:
        print(f"AI分类失败: {e}")
        return "需人工分类"

def process_files():
    os.makedirs(DIR_SOURCE, exist_ok=True)
    os.makedirs(DIR_MD, exist_ok=True)

    for root, dirs, files in os.walk(CONFIG["input_dir"]):
        for file in files:
            file_path = os.path.join(root, file)
            ext = os.path.splitext(file)[1].lower()
            
            if ext not in CONFIG["extensions"]:
                continue

            # 去重检查
            file_hash = get_file_hash(file_path)
            if file_hash in seen_hashes:
                print(f"[跳过重复] {file}")
                continue
            seen_hashes.add(file_hash)

            print(f"正在处理: {file} ...")

            # 转换为 Markdown
            try:
                result = md_converter.convert(file_path)
                md_content = result.text_content
            except Exception as e:
                print(f" 转换失败: {e}")
                md_content = ""

            # AI 分类
            category = ai_classify(md_content) if md_content else "未分类_其他"
            print(f" -> 分类结果: {category}")

            # 移动原始文件
            target_source_dir = os.path.join(DIR_SOURCE, category)
            os.makedirs(target_source_dir, exist_ok=True)
            shutil.copy2(file_path, os.path.join(target_source_dir, file))

            # 保存 Markdown 文件
            if md_content:
                target_md_dir = os.path.join(DIR_MD, category)
                os.makedirs(target_md_dir, exist_ok=True)
                
                md_filename = os.path.splitext(file)[0] + ".md"
                md_save_path = os.path.join(target_md_dir, md_filename)
                
                frontmatter = {
                    "created": "2026-03-08",
                    "original_path": file_path,
                    "tags": [category, "AutoOrganized"],
                    "hash": file_hash
                }
                
                with open(md_save_path, "w", encoding="utf-8") as f:
                    f.write("---\n")
                    yaml.dump(frontmatter, f, allow_unicode=True)
                    f.write("---\n\n")
                    f.write(md_content)

if __name__ == "__main__":
    process_files()
    print("整理完成！")
```
